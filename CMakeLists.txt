cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "macOS: Apple Silicon only")
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "macOS minimum version")
project(asr10emu C CXX)
if(APPLE)
  enable_language(OBJCXX)   # src/mac/menu_mac.mm (File / View menus)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(ASR10_GUI  "Build the SDL2 front end (asr10)" ON)
option(ASR10_MIDI "MIDI in/out through RtMidi" ON)
option(ASR10_STATIC_SDL "Link SDL2 statically (single executable, e.g. for Windows)" OFF)
option(ASR10_ASIO "ASIO audio driver support on Windows (Steinberg ASIO SDK files, GPLv3)" ON)

# --- Musashi 68000 core (pre-generated m68kops.c is included)
add_library(musashi STATIC
  musashi/m68kcpu.c musashi/m68kops.c musashi/m68kdasm.c musashi/softfloat/softfloat.c)
target_include_directories(musashi PUBLIC musashi musashi/softfloat src)

# --- machine core
add_library(asr10core STATIC
  src/asr10.cpp src/m68302.cpp src/duart.cpp src/fdc.cpp
  src/es5506.cpp src/es5510.cpp src/panel.cpp src/scsi.cpp)
target_include_directories(asr10core PUBLIC src musashi)
target_link_libraries(asr10core PUBLIC musashi)

# --- headless runner
add_executable(asr10-cli src/cli.cpp)
target_link_libraries(asr10-cli asr10core)

# --- SDL2 front end
if(APPLE)
  set(_bundled_sdl_default ON)
else()
  set(_bundled_sdl_default OFF)
endif()
option(ASR10_BUNDLED_SDL "Download SDL2 and link it statically (self-contained app)" ${_bundled_sdl_default})
if(ASR10_GUI)
  if(ASR10_BUNDLED_SDL)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
    set(SDL2_SOURCE_DIR "" CACHE PATH "Local SDL2 source tree (empty = download)")
    if(SDL2_SOURCE_DIR STREQUAL "")
      FetchContent_Declare(sdl2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.30.12/SDL2-2.30.12.tar.gz)
      FetchContent_MakeAvailable(sdl2)
    else()
      add_subdirectory(${SDL2_SOURCE_DIR} ${PROJECT_BINARY_DIR}/sdl2 EXCLUDE_FROM_ALL)
    endif()
    set(ASR10_STATIC_SDL ON)
  else()
    find_package(SDL2 REQUIRED)
  endif()
  add_executable(asr10 src/sdl_main.cpp src/vfd.cpp src/rack_ui.cpp src/ui_text.cpp src/native_menu.cpp src/audio_io.cpp src/settings.cpp third_party/rtaudio/RtAudio.cpp)
  if(APPLE)
    # ASR-10.app: ROMs and the OS disk in Contents/Resources/roms, icon, microphone permission
    set_target_properties(asr10 PROPERTIES
      MACOSX_BUNDLE TRUE
      OUTPUT_NAME "ASR-10"
      MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/src/mac/Info.plist.in
      MACOSX_BUNDLE_SHORT_VERSION_STRING "0.9"
      MACOSX_BUNDLE_BUNDLE_VERSION "0.9")
    target_sources(asr10 PRIVATE src/mac/asr10.icns src/mac/menu_mac.mm)
    target_link_libraries(asr10 "-framework Cocoa")
    set_source_files_properties(src/mac/asr10.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    file(GLOB _roms ${CMAKE_CURRENT_SOURCE_DIR}/roms/*.bin ${CMAKE_CURRENT_SOURCE_DIR}/roms/*.img)
    foreach(f ${_roms})
      target_sources(asr10 PRIVATE ${f})
      set_source_files_properties(${f} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/roms)
    endforeach()
  endif()
  target_compile_definitions(asr10 PRIVATE ASR_WITH_RTAUDIO)
  target_include_directories(asr10 PRIVATE third_party/rtaudio/include)
  find_package(Threads REQUIRED)
  target_link_libraries(asr10 Threads::Threads)
  # RtAudio back ends
  if(WIN32)
    target_compile_definitions(asr10 PRIVATE __WINDOWS_WASAPI__ __WINDOWS_DS__)
    if(MINGW)
      target_sources(asr10 PRIVATE src/win/wmcodecdsp_guids.c)
    else()
      target_link_libraries(asr10 wmcodecdsp)
    endif()
    target_link_libraries(asr10 ole32 winmm ksuser mfplat mfuuid dsound uuid comdlg32)
    if(ASR10_ASIO)
      target_compile_definitions(asr10 PRIVATE __WINDOWS_ASIO__)
      target_sources(asr10 PRIVATE third_party/rtaudio/include/asio.cpp third_party/rtaudio/include/asiodrivers.cpp
                                   third_party/rtaudio/include/asiolist.cpp third_party/rtaudio/include/iasiothiscallresolver.cpp)
    endif()
  elseif(APPLE)
    target_compile_definitions(asr10 PRIVATE __MACOSX_CORE__)
    target_link_libraries(asr10 "-framework CoreAudio" "-framework CoreFoundation" "-framework AudioToolbox")
  else()
    set(_rta_any OFF)
    find_package(ALSA)
    if(ALSA_FOUND)
      target_compile_definitions(asr10 PRIVATE __LINUX_ALSA__)
      target_link_libraries(asr10 ALSA::ALSA)
      set(_rta_any ON)
    endif()
    find_package(PkgConfig)
    if(PKG_CONFIG_FOUND)
      pkg_check_modules(PULSE IMPORTED_TARGET libpulse-simple)
      if(PULSE_FOUND)
        target_compile_definitions(asr10 PRIVATE __LINUX_PULSE__)
        target_link_libraries(asr10 PkgConfig::PULSE)
        set(_rta_any ON)
      endif()
      pkg_check_modules(JACK IMPORTED_TARGET jack)
      if(JACK_FOUND)
        target_compile_definitions(asr10 PRIVATE __UNIX_JACK__)
        target_link_libraries(asr10 PkgConfig::JACK)
        set(_rta_any ON)
      endif()
    endif()
    if(NOT _rta_any)
      message(WARNING "No ALSA/PulseAudio/JACK dev files: audio through SDL only")
      target_compile_definitions(asr10 PRIVATE __RTAUDIO_DUMMY__)
    endif()
  endif()
  if(WIN32)
    target_sources(asr10 PRIVATE src/win/asr10.rc)
  endif()
  if(ASR10_STATIC_SDL AND TARGET SDL2::SDL2-static)
    target_link_libraries(asr10 asr10core SDL2::SDL2-static)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(asr10 SDL2::SDL2main)
    endif()
  elseif(TARGET SDL2::SDL2)
    target_link_libraries(asr10 asr10core SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(asr10 SDL2::SDL2main)
    endif()
  else()
    target_include_directories(asr10 PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(asr10 asr10core ${SDL2_LIBRARIES})
  endif()

  if(ASR10_MIDI)
    target_sources(asr10 PRIVATE third_party/RtMidi.cpp)
    target_compile_definitions(asr10 PRIVATE ASR_WITH_RTMIDI)
    if(APPLE)
      target_compile_definitions(asr10 PRIVATE __MACOSX_CORE__)
      target_link_libraries(asr10 "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation")
    elseif(WIN32)
      target_compile_definitions(asr10 PRIVATE __WINDOWS_MM__)
      target_link_libraries(asr10 winmm)
    else()
      find_package(ALSA)
      if(ALSA_FOUND)
        target_compile_definitions(asr10 PRIVATE __LINUX_ALSA__)
        target_link_libraries(asr10 ALSA::ALSA pthread)
      else()
        message(WARNING "ALSA not found: MIDI disabled (install libasound2-dev)")
        target_compile_definitions(asr10 PRIVATE __RTMIDI_DUMMY__)
        target_link_libraries(asr10 pthread)
      endif()
    endif()
  endif()
endif()
